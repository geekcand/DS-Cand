-- CANNED TOOL: DUAL VOICE (MALE / FEMALE) -- MIGRATION v2 -- REQUIREMENTS: Clean Start schema + GSM Default v1.2 installed. -- RUN ON A BACKUP/STAGING DATABASE FIRST. Run in Supabase SQL Editor on the SAME project. -- This migration does NOT change any existing canned row or user profile. -- It adds a female variant and new admin/user RPCs. Old single-variant publish/add are disabled. -- Rollout: 1 SQL -> 2 new Admin -> 3 publish both xlsx -> 4 new Tool. BEGIN; DO $preflight$ BEGIN IF to_regclass('public.cand_fresh_library') IS NULL OR to_regclass('public.cand_fresh_access') IS NULL OR to_regclass('public.profiles') IS NULL THEN RAISE EXCEPTION 'CLEAN_START_REQUIRED: deploy the existing Clean Start schema first'; END IF; END $preflight$; ALTER TABLE public.cand_fresh_library ADD COLUMN IF NOT EXISTS content_female text; ALTER TABLE public.cand_fresh_library DROP CONSTRAINT IF EXISTS cand_fresh_library_female_nonblank; ALTER TABLE public.cand_fresh_library ADD CONSTRAINT cand_fresh_library_female_nonblank CHECK (content_female IS NULL OR length(btrim(content_female)) > 0); CREATE TABLE IF NOT EXISTS public.cand_voice_preferences ( user_id uuid PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE, voice text NOT NULL CHECK (voice IN ('male','female')), chosen_at timestamptz NOT NULL DEFAULT now(), updated_at timestamptz NOT NULL DEFAULT now(), updated_by uuid ); CREATE TABLE IF NOT EXISTS public.cand_voice_changes ( id bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY, user_id uuid NOT NULL, changed_by uuid NOT NULL, old_voice text, new_voice text NOT NULL CHECK (new_voice IN ('male','female')), changed_at timestamptz NOT NULL DEFAULT now() ); ALTER TABLE public.cand_voice_preferences ENABLE ROW LEVEL SECURITY; ALTER TABLE public.cand_voice_changes ENABLE ROW LEVEL SECURITY; REVOKE ALL ON public.cand_voice_preferences FROM PUBLIC, anon, authenticated; REVOKE ALL ON public.cand_voice_changes FROM PUBLIC, anon, authenticated; -- User may choose ONLY ONCE. Admin can correct mistakes using a separate secure function. CREATE OR REPLACE FUNCTION public.cand_voice_choose(p_voice text) RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path = '' AS $fn$ DECLARE v_uid uuid := auth.uid(); v_role text; v_status text; BEGIN IF v_uid IS NULL THEN RAISE EXCEPTION 'AUTH_REQUIRED'; END IF; SELECT p.role,p.status INTO v_role,v_status FROM public.profiles p WHERE p.id=v_uid; IF v_role <> 'user' OR v_status <> 'approved' THEN RAISE EXCEPTION 'APPROVED_AGENT_REQUIRED'; END IF; IF p_voice IS NULL OR p_voice NOT IN ('male','female') THEN RAISE EXCEPTION 'INVALID_VOICE'; END IF; IF EXISTS (SELECT 1 FROM public.cand_fresh_library WHERE content_female IS NULL) THEN RAISE EXCEPTION 'DUAL_CATALOG_NOT_PUBLISHED'; END IF; INSERT INTO public.cand_voice_preferences(user_id,voice,updated_by) VALUES(v_uid,p_voice,v_uid) ON CONFLICT (user_id) DO NOTHING; IF NOT FOUND THEN RAISE EXCEPTION 'VOICE_ALREADY_CHOSEN: contact Admin to correct'; END IF; RETURN jsonb_build_object('ok',true,'voice',p_voice); END $fn$; CREATE OR REPLACE FUNCTION public.cand_voice_admin_list() RETURNS TABLE(user_id uuid, voice text) LANGUAGE plpgsql STABLE SECURITY DEFINER SET search_path = '' AS $fn$ BEGIN IF NOT public.cand_fresh_admin() THEN RAISE EXCEPTION 'ADMIN_REQUIRED'; END IF; RETURN QUERY SELECT p.id, v.voice FROM public.profiles p LEFT JOIN public.cand_voice_preferences v ON v.user_id=p.id WHERE p.role='user'; END $fn$; CREATE OR REPLACE FUNCTION public.cand_voice_admin_set(p_user_id uuid,p_voice text) RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path = '' AS $fn$ DECLARE v_old text; v_actor uuid := auth.uid(); BEGIN IF NOT public.cand_fresh_admin() THEN RAISE EXCEPTION 'ADMIN_REQUIRED'; END IF; IF p_voice IS NULL OR p_voice NOT IN ('male','female') THEN RAISE EXCEPTION 'INVALID_VOICE'; END IF; IF NOT EXISTS(SELECT 1 FROM public.profiles WHERE id=p_user_id AND role='user') THEN RAISE EXCEPTION 'AGENT_NOT_FOUND'; END IF; SELECT voice INTO v_old FROM public.cand_voice_preferences WHERE user_id=p_user_id FOR UPDATE; INSERT INTO public.cand_voice_preferences(user_id,voice,updated_by,updated_at) VALUES(p_user_id,p_voice,v_actor,now()) ON CONFLICT(user_id) DO UPDATE SET voice=excluded.voice,updated_by=excluded.updated_by,updated_at=now(); IF v_old IS DISTINCT FROM p_voice THEN INSERT INTO public.cand_voice_changes(user_id,changed_by,old_voice,new_voice) VALUES(p_user_id,v_actor,v_old,p_voice); END IF; RETURN jsonb_build_object('ok',true,'user_id',p_user_id,'voice',p_voice); END $fn$; -- Atomic dual-file publication. Both variants are *paired* before the browser calls this. CREATE OR REPLACE FUNCTION public.cand_voice_publish(p_items jsonb,p_expected_count integer) RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path = '' AS $fn$ DECLARE v_count integer; v_all integer; v_gsm integer; v_distinct integer; BEGIN IF NOT public.cand_fresh_admin() THEN RAISE EXCEPTION 'ADMIN_REQUIRED'; END IF; IF p_items IS NULL OR jsonb_typeof(p_items)<>'array' THEN RAISE EXCEPTION 'INVALID_JSON_ARRAY'; END IF; v_count:=jsonb_array_length(p_items); IF v_count<2 OR v_count>10000 OR v_count IS DISTINCT FROM p_expected_count THEN RAISE EXCEPTION 'COUNT_MISMATCH expected=% actual=%',p_expected_count,v_count; END IF; IF EXISTS (SELECT 1 FROM jsonb_to_recordset(p_items) AS x( sheet_name text,title text,content text,content_female text,reference_url text, click_count integer,row_order integer,skills text) WHERE nullif(btrim(x.sheet_name),'') IS NULL OR nullif(btrim(x.title),'') IS NULL OR nullif(btrim(x.content),'') IS NULL OR nullif(btrim(x.content_female),'') IS NULL OR x.skills IS NULL OR x.skills NOT IN ('ALL','GSM') OR x.click_count IS NULL OR x.click_count < 0 OR x.row_order IS NULL OR x.row_order < 1 ) THEN RAISE EXCEPTION 'INVALID_PAIRED_CANNED_ITEM'; END IF; SELECT count(DISTINCT x.row_order), count(*) FILTER(WHERE skills='ALL'), count(*) FILTER(WHERE skills='GSM') INTO v_distinct,v_all,v_gsm FROM jsonb_to_recordset(p_items) AS x(row_order integer,skills text); IF v_distinct<>v_count OR v_all<1 OR v_gsm<1 THEN RAISE EXCEPTION 'INVALID_ORDER_OR_CLASSIFICATION'; END IF; IF NOT EXISTS(SELECT 1 FROM jsonb_to_recordset(p_items) AS x(row_order integer) WHERE x.row_order=v_count) THEN RAISE EXCEPTION 'NON_CONTIGUOUS_ROW_ORDER'; END IF; PERFORM pg_catalog.pg_advisory_xact_lock(816432101); -- WHERE is necessary for databases with pg_safeupdate enabled. DELETE FROM public.cand_fresh_library WHERE row_order > 0; INSERT INTO public.cand_fresh_library (sheet_name,title,content,content_female,reference_url,click_count,row_order,skills) SELECT x.sheet_name,x.title,x.content,x.content_female,x.reference_url,x.click_count,x.row_order,x.skills FROM jsonb_to_recordset(p_items) AS x(sheet_name text,title text,content text, content_female text,reference_url text,click_count integer,row_order integer,skills text) ORDER BY x.row_order; IF (SELECT count(*) FROM public.cand_fresh_library)<>v_count THEN RAISE EXCEPTION 'INCOMPLETE_DUAL_PUBLISH'; END IF; RETURN jsonb_build_object('ok',true,'total',v_count,'all_items',v_all,'gsm_items',v_gsm); END $fn$; -- Admin creates and edits both variants together; no single-sided edits. CREATE OR REPLACE FUNCTION public.cand_voice_add( p_sheet_name text,p_title text,p_content_male text,p_content_female text,p_skills text) RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path = '' AS $fn$ DECLARE v_order integer; BEGIN IF NOT public.cand_fresh_admin() THEN RAISE EXCEPTION 'ADMIN_REQUIRED'; END IF; IF nullif(btrim(p_sheet_name),'') IS NULL OR nullif(btrim(p_title),'') IS NULL OR nullif(btrim(p_content_male),'') IS NULL OR nullif(btrim(p_content_female),'') IS NULL OR p_skills IS NULL OR p_skills NOT IN('ALL','GSM') THEN RAISE EXCEPTION 'INVALID_CANNED_ITEM'; END IF; PERFORM pg_catalog.pg_advisory_xact_lock(816432101); SELECT coalesce(max(row_order),0)+1 INTO v_order FROM public.cand_fresh_library; INSERT INTO public.cand_fresh_library(sheet_name,title,content,content_female,skills,row_order) VALUES(btrim(p_sheet_name),btrim(p_title),p_content_male,p_content_female,p_skills,v_order); RETURN jsonb_build_object('ok',true,'row_order',v_order); END $fn$; CREATE OR REPLACE FUNCTION public.cand_voice_admin_catalog() RETURNS jsonb LANGUAGE plpgsql STABLE SECURITY DEFINER SET search_path = '' AS $fn$ BEGIN IF NOT public.cand_fresh_admin() THEN RAISE EXCEPTION 'ADMIN_REQUIRED'; END IF; RETURN (SELECT coalesce(jsonb_agg(jsonb_build_object( 'id',id,'sheet_name',sheet_name,'title',title,'content',content, 'content_female',content_female,'skills',skills,'reference_url',reference_url, 'row_order',row_order,'click_count',click_count) ORDER BY row_order),'[]'::jsonb) FROM public.cand_fresh_library); END $fn$; CREATE OR REPLACE FUNCTION public.cand_voice_update( p_id bigint,p_sheet_name text,p_title text,p_content_male text,p_content_female text, p_skills text,p_reference_url text) RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path = '' AS $fn$ BEGIN IF NOT public.cand_fresh_admin() THEN RAISE EXCEPTION 'ADMIN_REQUIRED'; END IF; IF nullif(btrim(p_sheet_name),'') IS NULL OR nullif(btrim(p_title),'') IS NULL OR nullif(btrim(p_content_male),'') IS NULL OR nullif(btrim(p_content_female),'') IS NULL OR p_skills IS NULL OR p_skills NOT IN ('ALL','GSM') THEN RAISE EXCEPTION 'INVALID_CANNED_ITEM'; END IF; PERFORM pg_catalog.pg_advisory_xact_lock(816432101); UPDATE public.cand_fresh_library SET sheet_name=btrim(p_sheet_name),title=btrim(p_title),content=p_content_male, content_female=p_content_female,skills=p_skills,reference_url=nullif(btrim(p_reference_url),'') WHERE id=p_id; IF NOT FOUND THEN RAISE EXCEPTION 'CANNED_NOT_FOUND_OR_REPUBLISHED'; END IF; RETURN jsonb_build_object('ok',true,'id',p_id); END $fn$; -- SERVER-SIDE variant filtering. Clients never receive the other variant's text. CREATE OR REPLACE FUNCTION public.cand_fresh_read() RETURNS jsonb LANGUAGE plpgsql STABLE SECURITY DEFINER SET search_path = '' AS $fn$ DECLARE v_uid uuid:=auth.uid();v_status text;v_role text; v_access text:='universal';v_voice text;v_total integer;v_all integer; BEGIN IF v_uid IS NULL THEN RAISE EXCEPTION 'AUTH_REQUIRED'; END IF; SELECT p.status,p.role INTO v_status,v_role FROM public.profiles p WHERE p.id=v_uid; IF v_role IS NULL OR v_status IS DISTINCT FROM 'approved' THEN RAISE EXCEPTION 'ACCOUNT_NOT_APPROVED'; END IF; IF v_role='admin' THEN v_access:='universal';v_voice:='male'; ELSIF v_role='user' THEN SELECT coalesce((SELECT a.skill_access FROM public.cand_fresh_access a WHERE a.user_id=v_uid),'universal') INTO v_access; SELECT voice INTO v_voice FROM public.cand_voice_preferences WHERE user_id=v_uid; ELSE RAISE EXCEPTION 'UNSUPPORTED_USER_ROLE'; END IF; SELECT count(*),count(*) FILTER(WHERE skills='ALL') INTO v_total,v_all FROM public.cand_fresh_library; IF v_total=0 THEN RAISE EXCEPTION 'CATALOG_NOT_PUBLISHED'; END IF; IF v_access='fixed' AND v_all=0 THEN RAISE EXCEPTION 'NO_ALL_ITEMS_FOR_FIXED'; END IF; -- Transition-safe: while Admin is preparing the dual workbook, legacy Tool keeps -- serving the existing male library. After dual publish, new Tool enforces selection. IF EXISTS (SELECT 1 FROM public.cand_fresh_library WHERE content_female IS NULL) THEN -- Never give a female-assigned agent male wording if the female catalog becomes incomplete. IF v_voice='female' THEN RAISE EXCEPTION 'DUAL_CATALOG_INCOMPLETE'; END IF; RETURN jsonb_build_object( 'requires_voice_selection',false,'voice','male','skill_access',v_access, 'total_catalog',v_total, 'visible_count',(SELECT count(*) FROM public.cand_fresh_library c WHERE v_access='universal' OR c.skills='ALL'), 'items',(SELECT coalesce(jsonb_agg(jsonb_build_object( 'sheet_name',c.sheet_name,'title',c.title,'content',c.content, 'reference_url',c.reference_url,'click_count',c.click_count, 'row_order',c.row_order,'skills',c.skills) ORDER BY c.row_order),'[]'::jsonb) FROM public.cand_fresh_library c WHERE v_access='universal' OR c.skills='ALL') ); END IF; IF v_voice IS NULL THEN RETURN jsonb_build_object('requires_voice_selection',true,'items','[]'::jsonb, 'skill_access',v_access,'visible_count',0,'total_catalog',v_total); END IF; RETURN jsonb_build_object( 'requires_voice_selection',false,'voice',v_voice,'skill_access',v_access, 'total_catalog',v_total, 'visible_count',(SELECT count(*) FROM public.cand_fresh_library c WHERE v_access='universal' OR c.skills='ALL'), 'items',(SELECT coalesce(jsonb_agg(jsonb_build_object( 'sheet_name',c.sheet_name,'title',c.title, 'content',CASE WHEN v_voice='female' THEN c.content_female ELSE c.content END, 'reference_url',c.reference_url,'click_count',c.click_count, 'row_order',c.row_order,'skills',c.skills) ORDER BY c.row_order),'[]'::jsonb) FROM public.cand_fresh_library c WHERE v_access='universal' OR c.skills='ALL') ); END $fn$; -- Disable obsolete one-file writes (avoid accidental loss of female content).
